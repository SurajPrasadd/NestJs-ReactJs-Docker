            }

            navItems.forEach(nav => nav.classList.remove('active'));
            this.classList.add('active');
        });
    });

    // Table wrapper right scroll for mobile
    function autoScrollTableRightIfMobile() {
        const wrapper = document.getElementById('tableWrapper');
        if (!wrapper) return;
        const isMobile = window.innerWidth <= 768;
        if (isMobile) {
            // Scroll to the far right to reveal later columns
            wrapper.scrollLeft = wrapper.scrollWidth;
        }
    }

    // Initial Render
    renderTable(leadsData);
    updatePaginationInfo();
    autoScrollTableRightIfMobile();

    // Adjust on resize
    window.addEventListener('resize', autoScrollTableRightIfMobile);
</script>
